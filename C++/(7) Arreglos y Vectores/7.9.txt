#include <iostream>
using std::cout;
using std::cin;
using std::endl;
#include <conio.h>

int main()
{
	int t[2][3];	//Inciso a).
	t[0][1] = 0;	//Inciso g).

	t[0][0] = 0;////////////////////////////////
	t[0][2] = 0;
	t[1][0] = 0;		//Inciso h).
	t[1][1] = 0;
	t[1][2] = 0;///////////////////////////////

	//////////////////////////////////////////////////////////////////////
	for (int i = 0; i < 2; i++)
		for (int j = 0; j < 3; j++)		//Inciso i).
			t[i][j] = 0;
	/////////////////////////////////////////////////////////////////////

	//////////////////////////////////////////////////////////////////////////////////
	for (int k = 0; k<2; k++)
		for (int l = 0; l < 3; l++)
		{
			cout << "Ingrese el valor de t[" << k << "][" << l << "]:";	//Inciso j).
			cin >> t[k][l];
		}
	cout << endl;
	/////////////////////////////////////////////////////////////////////////////////

	////////////////////////////////////////////////////////////////////////
	int masPequenio = 0;
	masPequenio = t[0][0];
	for (int m = 0; m < 2; m++)
	{
		for (int n = 0; n < 3; n++)				//Inciso k).
		{
			if (masPequenio > t[m][n])
				masPequenio = t[m][n];
		}
	}
	cout << "El elemento mas pequenio del arreglo bidimensional es " << masPequenio << endl;
	////////////////////////////////////////////////////////////////////////

	///////////////////////////////////////////////////////////////////////
	cout << "Fila 0: ";
	for (int o = 0; o < 3; o++)
		cout << t[0][o] << ", ";				//Inciso l).
	cout << endl;
	//////////////////////////////////////////////////////////////////////

	//////////////////////////////////////////////////////////////////////
	cout << "Columna 3: ";
	for (int p = 0; p < 2; p++)
		cout << t[p][2] << ", ";				//Inciso m).
	cout << endl;
	//////////////////////////////////////////////////////////////////////

	//////////////////////////////////////////////////////////////////////
	cout << "Columna		[0]		[1]		[2]" << endl;
	for (int q = 0; q < 2; q++)
	{
		cout <<"Fila"<< "[" << q << "]";
		for (int r = 0; r < 3; r++)				//Inciso n).
			cout << "		 " << t[q][r];
		cout << endl;
	}
	///////////////////////////////////////////////////////////////////////
	_getch();
}
